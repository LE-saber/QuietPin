cmake_minimum_required(VERSION 3.20)
project(QuietPin VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "QuietPin requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(quietpin_core STATIC src/Config.cpp src/WindowOps.cpp src/PinPlacement.cpp)
target_include_directories(quietpin_core PUBLIC src)
target_compile_definitions(quietpin_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(quietpin_core PUBLIC user32 advapi32 dwmapi shell32 ole32)
add_executable(QuietPin WIN32 src/main.cpp src/PinOverlay.cpp src/PinController.cpp src/WindowEventMonitor.cpp resources/app.rc)
target_include_directories(QuietPin PRIVATE resources)
target_link_libraries(QuietPin PRIVATE quietpin_core comctl32 gdi32 comdlg32 oleacc oleaut32 uuid wtsapi32)
if(MSVC)
  target_compile_options(quietpin_core PRIVATE /W4 /utf-8)
  target_compile_options(QuietPin PRIVATE /W4 /utf-8)
  target_link_options(QuietPin PRIVATE /MANIFEST:NO)
else()
  target_compile_options(quietpin_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(QuietPin PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(QuietPin PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_compile_options(QuietPin PRIVATE $<$<CONFIG:Release>:-Os> $<$<CONFIG:Release>:-ffunction-sections> $<$<CONFIG:Release>:-fdata-sections>)
  target_link_options(QuietPin PRIVATE $<$<CONFIG:Release>:-Wl,--gc-sections> $<$<CONFIG:Release>:-Wl,--strip-all>)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(quietpin_tests tests/core_tests.cpp)
  target_link_libraries(quietpin_tests PRIVATE quietpin_core)
  add_executable(quietpin_integration tests/integration.cpp)
  target_link_libraries(quietpin_integration PRIVATE quietpin_core)
  add_executable(quietpin_pin_integration tests/pin_integration.cpp)
  target_link_libraries(quietpin_pin_integration PRIVATE quietpin_core oleacc oleaut32 uuid)
  if(NOT MSVC)
    target_link_options(quietpin_tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(quietpin_integration PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(quietpin_integration PRIVATE -municode)
    target_link_options(quietpin_pin_integration PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME core COMMAND quietpin_tests)
  add_test(NAME integration COMMAND quietpin_integration $<TARGET_FILE:QuietPin>)
  add_test(NAME pin_integration COMMAND quietpin_pin_integration $<TARGET_FILE:QuietPin>)
  set_tests_properties(integration pin_integration PROPERTIES TIMEOUT 35 RUN_SERIAL TRUE)
endif()
